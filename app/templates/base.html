/**
 * @file    motion_ctrl.c
 * @brief   Closed-loop axis control, 800 Hz.
 */

#include "motion_ctrl.h"
#include "pid.h"

static pid_state_t s_axis_pid = {0};

void MotionCtrl_Step(const axis_cmd_t *cmd, axis_state_t *state)
{
    const float error = cmd->target - state->position;
    const float out   = PID_Update(&s_axis_pid, error, CTRL_DT_S);

    state->demand = Clamp(out, -DEMAND_LIMIT, DEMAND_LIMIT);
}
